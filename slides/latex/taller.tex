        \begin{center}
        \begin{tikzpicture}[
            box/.style={draw=white, thick, rounded corners, minimum width=2.1cm, minimum height=1.4cm, align=center, text=white, font=\small, inner sep=2pt},

            \draw[arr] (verif.south) to[out=-65,in=-115]
                node[below,font=\tiny,text=white,pos=0.5,yshift=-0.35cm]{ajusta el prompt y repite}

    \section{Plotly guiado}

    \begin{frame}{Objetivo del bloque}
        Generar tus primeras gráficas con Plotly, guiado paso a paso, usando
        datos abiertos reales.

        \vspace{1em}
        \textbf{Tú} rediges los prompts. \textbf{La IA} escribe y corre el código.
        \textbf{Ambos} verifican el resultado.
    \end{frame}

    \begin{frame}{Qué vamos a construir}
        \begin{itemize}
            \item Gráfica de \textbf{barras} — comparar categorías
            \item Gráfica de \textbf{líneas} — ver una tendencia en el tiempo
            \item \textbf{Mapa} — ver un dato distribuido geográficamente
        \end{itemize}
    \end{frame}

    \begin{frame}{¿Dónde voy a ver la gráfica?}
        \begin{itemize}
            \item \textbf{Se abre sola} — si el script usa \texttt{fig.show()}, aparece en tu navegador automáticamente
            \item \textbf{Se guarda como HTML} — \texttt{fig.write\_html(...)}, la abres con doble clic (sigue siendo interactiva)
            \item \textbf{Se guarda como imagen} — \texttt{fig.write\_image(...)}, ya no es interactiva
        \end{itemize}

        \vspace{0.8em}
        Si el script corre y no ves nada, pide: \textit{“ábrela en el navegador”}.
    \end{frame}

    \begin{frame}{De dónde salen los datos}
        Vamos a trabajar con \textbf{datos abiertos del INEGI} (Censo de
        Población y Vivienda 2020) — datos reales, no de ejemplo:
        \begin{itemize}
            \item \texttt{poblacion\_por\_estado.csv} — para barras y mapa
            \item \texttt{poblacion\_mexico\_historica.csv} — para líneas
        \end{itemize}

        Antes de dar por buena una gráfica generada por IA, revisa:
        \begin{itemize}
            \item ¿Los ejes muestran lo que dicen mostrar (unidades, escala)?
            \item ¿Los números coinciden con lo que esperarías del dataset?
            \item ¿La IA usó el archivo real, o inventó valores de ejemplo?
        \end{itemize}
    \end{frame}

    \section{De Plotly a Dash}

    \begin{frame}{¿Qué es Dash?}
        Un framework de Python que sirve tus gráficas de Plotly como una
        \textbf{aplicación web interactiva}, corriendo en un puerto local
        (\texttt{localhost}). Plotly genera la gráfica; Dash la convierte en
        dashboard.

        \vspace{0.6em}
        \begin{center}
            \includegraphics[width=9.5cm]{images/example-dash.png}
        \end{center}
        {\tiny\textit{Ejemplo real: dashboard con filtro interactivo, generado para esta demo.}}
    \end{frame}

    \begin{frame}{Qué vamos a construir}
        Un dashboard que reúne las gráficas de los bloques anteriores en una
        sola página, con al menos un control interactivo (por ejemplo, un
        filtro) que actualiza la gráfica.
    \end{frame}

    \begin{frame}{Cómo lo vamos a hacer}
        \begin{enumerate}
            \item Partimos del código Plotly que ya generamos
            \item Le pedimos a la IA que lo integre en una app Dash
            \item Corremos la app y la abrimos en el navegador (\texttt{localhost})
            \item Verificamos que el control interactivo funcione
        \end{enumerate}
    \end{frame}

    \section{Buenas prácticas y cierre}

    \begin{frame}{Buenas prácticas de uso responsable}
        \begin{itemize}
            \item Verifica siempre — la IA puede inventar datos con total confianza
            \item No compartas datos sensibles o confidenciales con herramientas de IA
            \item Documenta qué prompts usaste para poder reproducir tu trabajo
            \item La IA acelera el trabajo técnico, no reemplaza tu criterio
        \end{itemize}
    \end{frame}

    \begin{frame}{Privacidad: cuidado con datos de investigación}
        Como investigador, lo que subes a una IA puede incluir datos que no
        son solo tuyos.
        \begin{itemize}
            \item Antes de subir un dataset, revisa si tiene datos personales identificables (nombres, correos, ubicaciones, folios)
            \item Anonimiza o agrega los datos sensibles antes de compartirlos con cualquier IA
            \item Revisa la política de privacidad de la herramienta — algunas usan tus conversaciones para entrenar modelos por defecto (busca la opción de desactivarlo)
            \item Consulta las políticas de tu institución o comité de ética antes de usar datos de investigación con IA
        \end{itemize}
    \end{frame}

    \begin{frame}{¿Cuándo desconfiar de un resultado?}
        \begin{itemize}
            \item Los números son "demasiado perfectos" o redondos
            \item La gráfica no cambia aunque cambies el dataset de entrada
            \item La IA no puede explicar de dónde salió un valor específico
            \item El resultado contradice algo que ya sabes del tema
        \end{itemize}
    \end{frame}
